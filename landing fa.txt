  <script>
    document.addEventListener("DOMContentLoaded", function () {
      const config = {
        intervalo: 5,     // Cada 5 segundos aparece una notificación
        duracion: 3.5,    // Permanece visible 3.5 segundos
        productName: "🎁 Premio PAGADO$",
        productImage: "https://imgfz.com/i/gG8bTM6.jpeg"
      };

      const nombres = ["Juan Gómez", "María López", "Carlos Pérez", "Laura Castro", "Exe Bosch", "Matías Silva", "Sofía Fernández"];
      const ciudades = ["Jujuy", "Córdoba", "Tucumán", "Rosario", "Buenos Aires", "Mendoza", "Salta"];

      function getRandom(arr) {
        return arr[Math.floor(Math.random() * arr.length)];
      }

      function getTimeAgo() {
        const h = Math.floor(Math.random() * 2);
        const m = Math.max(1, Math.floor(Math.random() * 55));
        return h === 0 ? `hace ${m} minutos` : `hace ${h} hora`;
      }

      function showNotification() {
        // Remover notificación previa si aún existe
        const existing = document.querySelector(".xpo-notification-wrap");
        if (existing) existing.remove();

        const nombre = getRandom(nombres);
        const ciudad = getRandom(ciudades);
        const tiempo = getTimeAgo();

        const noti = document.createElement("div");
        noti.className = "xpo-notification-wrap";
        noti.innerHTML = `
          <div class="xpo-img-wrap">
            <img src="${config.productImage}" alt="Premio" onerror="this.src='https://i.imgur.com/GCxNjBq.jpeg'">
          </div>
          <div class="xpo-col">
            <div class="xpo-notification-name">${nombre}</div>
            <div class="xpo-notification-product">${config.productName}</div>
            <div class="xpo-location">📍 Desde ${ciudad} • ${tiempo}</div>
          </div>
        `;
        document.body.appendChild(noti);
        
        requestAnimationFrame(() => {
          setTimeout(() => noti.classList.add("show"), 50);
        });

        setTimeout(() => {
          noti.classList.remove("show");
          setTimeout(() => noti.remove(), 600);
        }, config.duracion * 1000);
      }

      // Primera notificación tras 1.5s de carga
      setTimeout(showNotification, 1500);
      setInterval(showNotification, config.intervalo * 1000);
    });
  </script>
